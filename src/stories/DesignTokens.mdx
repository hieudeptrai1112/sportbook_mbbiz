import globalTokens from './tokens-global.json';
import aliasTokens from './tokens-alias.json';

export const meta = { title: 'Design Token' };

export const collections = [
  { id: 'global', title: 'Global Tokens', list: globalTokens },
  { id: 'alias', title: 'Alias Tokens', list: aliasTokens }
];

export const isColor = (t) => t.t === 'COLOR' && typeof t.v === 'string';

export const renderTable = (list) => (
  <table className="token-table">
    <thead>
      <tr>
        <th>Token</th>
        <th>Value</th>
        <th>Type</th>
        <th>Collection</th>
      </tr>
    </thead>
    <tbody>
      {list.map((t) => (
        <tr key={t.n}>
          <td>
            {isColor(t) && <span className="token-swatch" style={{ background: t.v }} />}
            {t.n}
          </td>
          <td>{String(t.v)}</td>
          <td>{t.t}</td>
          <td>{t.c}</td>
        </tr>
      ))}
    </tbody>
  </table>
);

export const renderByCollection = (filter) => (
  <div className="token-section">
    {collections.map((collection) => {
      const list = collection.list.filter(filter);
      if (!list.length) return null;
      return (
        <div key={collection.id} className="token-section">
          <h4>{collection.title}</h4>
          {renderTable(list)}
        </div>
      );
    })}
  </div>
);
