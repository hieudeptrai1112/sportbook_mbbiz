import globalTokens from './tokens-global.json';
import aliasTokens from './tokens-alias.json';

export const meta = { title: 'Design Tokens/Catalog' };

# Token Catalog

This page lists every token currently available in the library.

export const isColor = (t) => t.t === 'COLOR' && typeof t.v === 'string';
export const groups = [
  { id: 'color/primitive', title: 'Color - Primitive', filter: (t) => t.n.startsWith('color/primitive/') },
  { id: 'color/semantic', title: 'Color - Semantic', filter: (t) => t.n.startsWith('color/semantic/') },
  { id: 'spacing', title: 'Spacing', filter: (t) => t.n.startsWith('spacing/') },
  { id: 'radius', title: 'Radius', filter: (t) => t.n.startsWith('radius/') },
  { id: 'padding', title: 'Padding', filter: (t) => t.n.startsWith('padding/') },
  { id: 'font', title: 'Font', filter: (t) => t.n.startsWith('font/') || t.n.startsWith('typography/') },
  { id: 'size', title: 'Size', filter: (t) => t.n.startsWith('width/') || t.n.startsWith('iconsize/') || t.n.startsWith('stroke/') },
  { id: 'effect', title: 'Effect', filter: (t) => t.n.startsWith('effect/') }
];

export const collections = [
  { id: 'global', title: 'Global Tokens', list: globalTokens },
  { id: 'alias', title: 'Alias Tokens', list: aliasTokens }
];

export const renderTable = (list) => (
  <table className="token-table">
    <thead>
      <tr>
        <th>Token</th>
        <th>Value</th>
        <th>Type</th>
        <th>Collection</th>
      </tr>
    </thead>
    <tbody>
      {list.map((t) => (
        <tr key={t.n}>
          <td>
            {isColor(t) && <span className="token-swatch" style={{ background: t.v }} />}
            {t.n}
          </td>
          <td>{String(t.v)}</td>
          <td>{t.t}</td>
          <td>{t.c}</td>
        </tr>
      ))}
    </tbody>
  </table>
);

<div className="token-section">
  <strong>Total tokens:</strong> {globalTokens.length + aliasTokens.length}
</div>

{collections.map((collection) => (
  <section key={collection.id} className="token-section">
    <h2>{collection.title}</h2>
    {groups.map((group) => {
      const list = collection.list.filter(group.filter);
      if (!list.length) return null;
      return (
        <div key={`${collection.id}-${group.id}`} className="token-section">
          <h3>{group.title}</h3>
          {renderTable(list)}
        </div>
      );
    })}
  </section>
))}
